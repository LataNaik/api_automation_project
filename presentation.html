How It Works

def test_create_individual():
    # 1. Authenticate
    token = get_token()

    # 2. Create API client
    client = APIClient(service="individual", token=token)

    # 3. Load payload template
    payload = load_json_payload("individual_create.json")

    # 4. Inject dynamic data
    payload["Individual"]["id"] = str(uuid.uuid4())

    # 5. Make API call
    response = client.post("/v1/_create", json=payload)

    # 6. Validate response
    assert response.status_code in [200, 202]

    # 7. Save for future tests
    save_id("individual_id", payload["Individual"]["id"])

Reusable Components

# Generic search helper - works for ANY service
def search_entity(client, endpoint, id_value,
                  response_key, save_name):
    """One function for all search operations"""
    payload = {
        response_key: {"id": [id_value]},
        "RequestInfo": build_request_info()
    }
    response = client.post(endpoint, json=payload)
    assert response.status_code == 200
    return response.json()

# Use for any service:
search_entity(client, "/individual/v1/_search", id, "Individual", "ind")
search_entity(client, "/facility/v1/_search", id, "Facility", "fac")
search_entity(client, "/product/v1/_search", id, "Product", "prod")

Quick Start

# 1. Clone and setup
git clone <repository-url>
cd api_automation_project
python3 -m venv .venv
source .venv/bin/activate

# 2. Install dependencies
pip install -r requirements.txt

# 3. Configure
cp .env.example .env
# Edit .env with your credentials

# 4. Run tests
pytest

# 5. Generate reports
pytest --html=reports/report.html --alluredir=reports/allure

Running Tests

# Run all tests
pytest

# Run specific service
pytest tests/test_individual_service.py

# Verbose mode
pytest -v

# With HTML report
pytest --html=reports/report.html

# Stop on first failure
pytest -x

# Run in parallel (with pytest-xdist)
pytest -n auto
